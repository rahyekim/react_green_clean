-- 3. 권한 부여 (생성한 계정 이름인 BLG_USR로 통일)
GRANT CONNECT TO "BLG_USR";
GRANT RESOURCE TO "BLG_USR";

-- 4. 테이블스페이스 및 용량 무제한 설정 (문법 완성)

ALTER USER "BLG_USR" QUOTA UNLIMITED ON "USERS";

ALTER USER BLG_USR DEFAULT TABLESPACE USERS QUOTA UNLIMITED ON USERS;

CONSTRAINT PK_BLG_CONTENT PRIMARY KEY(BLG_CONT_SEQ)
);  //수동 AUTO_INCREMENT

----[contacts]

create TABLE if NOT EXISTS contacts (
id INT AUTO_INCREMENT PRIMARY KEY,
NAME VARCHAR(50) NOT NULL,
phone VARCHAR(30) NOT NULL,
email VARCHAR(100) NOT NULL,
message TEXT NOT NULL,
is_replied TINYINT(1) DEFAULT 0,
action_memo VARCHAR(255) DEFAULT NULL,
created_at TIMESTAMP DEFAULT current_timestamp

);
